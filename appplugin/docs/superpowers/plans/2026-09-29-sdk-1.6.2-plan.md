private val warmedAll = AtomicReference<CountDownLatch?>(null)

fun start(base: Context) {
    val baseLatch = CountDownLatch(1)
    if (!warmed.compareAndSet(null, baseLatch)) return
    val allLatch = CountDownLatch(1).also { warmedAll.set(it) }

private fun warmAll(base: Context, baseLatch: CountDownLatch, allLatch: CountDownLatch) {

internal fun awaitAll(timeoutMs: Long): Long {

/** Thân của await() 1.6.1 (`:116-138`), nhãn log là [label]: chữ của await() giữ nguyên ("prefs warmup waited=" …). */
private fun awaitLatch(latch: CountDownLatch?, timeoutMs: Long, label: String): Long {
    if (latch == null) {
        Log.d(LogUtil.TAG, "$label not started in this process, main reads as before")
        return 0
    }
    val startedAt = System.nanoTime()
    val done = try {
        latch.await(timeoutMs, TimeUnit.MILLISECONDS)
    } catch (e: InterruptedException) {
        Thread.currentThread().interrupt()
        val waitedMs = TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - startedAt)
        Log.w(LogUtil.TAG, "$label wait interrupted after $waitedMs ms, main reads as before")
        return waitedMs
    }
    val waitedMs = TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - startedAt)
    if (done) Log.d(LogUtil.TAG, "$label waited=$waitedMs ms (timeout=$timeoutMs ms)")
    else Log.w(LogUtil.TAG, "$label timeout after $timeoutMs ms, main reads as before")
    return waitedMs
}
