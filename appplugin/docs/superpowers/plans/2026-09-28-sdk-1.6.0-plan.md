## Global Constraints

Nguyên văn từ brief (tiếng Anh, giữ đúng chữ; kế thừa 1.5.0 và bổ sung của brief 1.6.0):

> - no public/protected API change unless a task is explicitly marked as owner-approved (a task that needs one goes under "Blocked on owner" instead of being planned);
> - no push/tag/rebase/merge;
> - one implementer at a time; a fresh implementer per task, one review per task, red→green logs, and a ledger;
> - red log before green for behavior changes (pure moves may instead prove equivalence by the full suite + api_dump --check + release_gate.sh);
> - every app request keeps exactly one terminal callback;
> - no main-thread disk I/O (StrictMode G2 = 0);
> - consumer apps in ~/AndroidStudioProjects are read-only;
> - emulator: G9 launches via `am start`, not monkey; screenshot before any close tap;
> - do not plan any real-device step that changes settings other than rotation, and rotation must be read before and restored after.

```kotlin
package com.app.plugin.ads.core

/** Gọi tới mọi callback của cùng một lượt load, mỗi cái đúng một lần; một callback ném không chặn cái kia (Điều 7). O7 (a). */
internal class FanOutLoadCb(targets: List<AdsLoadCb>) : AdsLoadCb {
    val targets: List<AdsLoadCb> = targets.distinct()
    override fun onLoaded(placement: String, originPl: String, adId: String) =
        targets.forEach { t -> runCatching { t.onLoaded(placement, originPl, adId) }.onFailure { LogUtil.logE("ads load cb onLoaded threw: ${it.message}") } }
    override fun onLoadFailed(placement: String, originPl: String, adId: String, error: String) =
        targets.forEach { t -> runCatching { t.onLoadFailed(placement, originPl, adId, error) }.onFailure { LogUtil.logE("ads load cb onLoadFailed threw: ${it.message}") } }
}

/** Gắn [cb] vào lượt đang chạy của [plcf]: rỗng → gán; khác → fan-out (phẳng); trùng/null → không đổi. Setter của cbLoad ghi lượt (Task 15 1.5.0) nên isLoadCallbackCurrent vẫn đúng. Nằm trong `internal object`: hàm top-level `internal` không mangle, sinh khối `LoadCallbacksKt` → facade FAIL (review Task 0 I2). */
internal object LoadCallbacks {
    fun attachLoadCallback(plcf: AdsPlacementConfig, cb: AdsLoadCb?) {
        if (cb == null) return
        val cur = plcf.cbLoad
        plcf.cbLoad = when {
            cur == null -> cb
            cur === cb || (cur is FanOutLoadCb && cb in cur.targets) -> return
            cur is FanOutLoadCb -> FanOutLoadCb(cur.targets + cb)
            else -> FanOutLoadCb(listOf(cur, cb))
        }
    }
}

/** Báo `cf not` đồng bộ đúng một lần; lời gọi lồng (app load lại ngay trong onLoadFailed của cùng placement) được post sang vòng kế để không đệ quy đồng bộ. */
internal class CfNotDispatcher(private val tag: String) {
    private val dispatching = java.util.Collections.synchronizedSet(HashSet<String>())
    private val main = Handler(Looper.getMainLooper())
    fun fail(placement: String, realPl: String, cb: AdsLoadCb?) {

            return

        try { cb.onLoadFailed(placement, realPl, "", "cf not") } finally { dispatching.remove(realPl) }
    }
}
```

```kotlin
// core/LoadCallbacks.kt (thêm)
/** Callback của lượt đang chạy lúc host bị huỷ (review Task 4 I1): chuyển đúng một callback kết thúc rồi bỏ tham chiếu tới
 *  callback của app (thường giữ Activity đã huỷ); lời gọi sau đó (onLoaded muộn của banner, GMA refresh rect) bị bỏ như
 *  callback rảnh của Task 4. */
internal class OneShotLoadCb(inner: AdsLoadCb) : AdsLoadCb {
    @Volatile private var inner: AdsLoadCb? = inner
    fun holds(cb: AdsLoadCb): Boolean = inner?.let { LoadCallbacks.holds(it, cb) } ?: false
    override fun onLoaded(placement: String, originPl: String, adId: String) {
        val t = inner ?: return
        inner = null
        t.onLoaded(placement, originPl, adId)
    }
    override fun onLoadFailed(placement: String, originPl: String, adId: String, error: String) {
        val t = inner ?: return
        inner = null
        t.onLoadFailed(placement, originPl, adId, error)
    }
}

// trong internal object LoadCallbacks (thêm); attachLoadCallback thay nhánh
// `cur === cb || (cur is FanOutLoadCb && cb in cur.targets) -> return` bằng `holds(cur, cb) -> return`
fun holds(cur: AdsLoadCb?, cb: AdsLoadCb): Boolean =
    cur === cb || (cur is OneShotLoadCb && cur.holds(cb)) || (cur is FanOutLoadCb && cur.targets.any { holds(it, cb) })
```

```kotlin
 *  bỏ tham chiếu. Gán qua setter: `cbLoadCycle` = lượt đang chạy — vòng đang chạy vẫn hiện hành; sau `load timeout`
 *  (`loadCycle` tăng) `onLoaded` muộn vẫn bị chặn như trước. Trả số callback đã bọc. */
internal fun wrapInFlightLoadCallbacks(): Int {

        val cur = cf.cbLoad ?: continue
        if (cf.isLoading && cur !is OneShotLoadCb) {
            cf.cbLoad = OneShotLoadCb(cur)
            wrapped++
        }
    }
    return wrapped
}
```

```kotlin
internal object TelemetryConsentGate {
    enum class Decision { PASS, HOLD, DROP }
    const val HOLD_CAP = 50                       // first_open, session_start, fir_rm_fet_*, config change… trước consent là ≤ 10; 50 là trần an toàn
    const val KEY_PENDING = "telemetry_pre_consent_pending"   // PreferenceHelper: số event đang giữ (ghi trên IO), đọc ở lần mở kế → pre_consent_lost

    /** Tầng dùng được chưa: GRANTED/PARTIAL hoặc ngoài GDPR → PASS; DENIED → DROP; UNKNOWN (trong GDPR hoặc chưa biết) → HOLD. Cùng luật ConsentRules.tierAllows. */
    fun decide(tier: ConsentTier, gdprApplies: Boolean?): Decision = when {
        ConsentRules.tierAllows(tier, gdprApplies) -> Decision.PASS
        tier == ConsentTier.DENIED -> Decision.DROP
        else -> Decision.HOLD
    }
    fun install(context: Context)        // giữ appContext; ConsentState.onChange { reevaluate() } (đăng ký một lần; resetForTest gỡ)
    fun submitOrHold(item: LogEventQueue.Item, submit: (LogEventQueue.Item) -> Unit)   // @Synchronized; gọi trên IO
    fun reevaluate()                     // @Synchronized: PASS → xả theo thứ tự; DROP → bỏ, item.onOutcome?.invoke(DROPPED), đếm; ghi KEY_PENDING trên IO
    fun consumePendingLostForSessionStart(): Int   // đọc KEY_PENDING (prefs đã warm-up), đặt lại 0 trên IO
    // bộ đếm cho log/report: held, flushed, dropped (DENIED), evicted (quá trần)
}
```
